new ToastBuilder("Uploading...")
  .withDuration(6000)
  .withProgress({ mode: "drain", color: "#28a6f5", label: "Upload progress" })
  .withOnClose((reason) => {
    if (reason !== "timeout") return;
    new ToastBuilder("Upload complete!").withSeverity(ToastSeverity.SUCCESS).show();
  })
  .show();
